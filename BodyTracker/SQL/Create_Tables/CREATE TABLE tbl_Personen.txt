CREATE TABLE `tbl_Personen` (
	`person_id` INT(11) NOT NULL AUTO_INCREMENT,
	`person_first_name` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`person_last_name` VARCHAR(50) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`person_birth_date` DATE NULL DEFAULT NULL,
	`person_height` FLOAT NULL DEFAULT NULL,
	PRIMARY KEY (`person_id`) USING BTREE
)
COLLATE='utf8_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=2
;
