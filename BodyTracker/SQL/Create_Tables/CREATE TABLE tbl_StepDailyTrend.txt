CREATE TABLE `tbl_StepTrend` (
	`step_trend_id` BIGINT(20) NOT NULL AUTO_INCREMENT,
	`person_id` INT(11) NOT NULL,
	`binning_data` VARCHAR(500) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`update_at` TIMESTAMP NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
	`create_at` DATETIME NULL DEFAULT NULL,
	`source_package` VARCHAR(255) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`source_type` INT(11) NULL DEFAULT NULL,
	`step_count_current` INT(11) NULL DEFAULT NULL,
	`step_count_initial` INT(11) NULL DEFAULT NULL,
	`speed_current` DOUBLE NULL DEFAULT NULL,
	`speed_initial` DOUBLE NULL DEFAULT NULL,
	`distance_current` DOUBLE NULL DEFAULT NULL,
	`distance_initial` DOUBLE NULL DEFAULT NULL,
	`calories_current` DOUBLE NULL DEFAULT NULL,
	`calories_initial` DOUBLE NULL DEFAULT NULL,
	`device_uuid` VARCHAR(255) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`package_name` VARCHAR(255) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`data_uuid` CHAR(36) NULL DEFAULT NULL COLLATE 'utf8_general_ci',
	`record_date` DATETIME NOT NULL,
	PRIMARY KEY (`step_trend_id`) USING BTREE,
	UNIQUE INDEX `UNIQUE KEY` (`data_uuid`) USING BTREE,
	INDEX `KEY` (`person_id`) USING BTREE,
	CONSTRAINT `fk_step_trend` FOREIGN KEY (`person_id`) REFERENCES `tbl_Persons` (`person_id`) ON UPDATE CASCADE ON DELETE CASCADE
)
COLLATE='utf8_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=4525
;
