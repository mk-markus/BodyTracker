ALTER TABLE tbl_StepTrend
    -- IDs & Metadaten
    CHANGE COLUMN StepDailyTrendID step_trend_id BIGINT(20) NOT NULL AUTO_INCREMENT,
    CHANGE COLUMN PersonID_FK person_id BIGINT(20) NOT NULL,
    CHANGE COLUMN day_time record_date DATETIME NOT NULL,
    CHANGE COLUMN binning_data binning_data VARCHAR(500) NULL,
    CHANGE COLUMN source_pkg_name source_package VARCHAR(255) NULL,
    CHANGE COLUMN source_type source_type INT(11) NULL,
    CHANGE COLUMN device_uuid device_uuid VARCHAR(255) NULL,
    CHANGE COLUMN pkg_name package_name VARCHAR(255) NULL,
    CHANGE COLUMN data_uuid data_uuid CHAR(36) NULL,
    CHANGE COLUMN create_time create_at DATETIME NULL,

    -- Step Count
    CHANGE COLUMN count step_count_current INT(11) NULL,
    ADD COLUMN step_count_initial INT(11) NULL AFTER step_count_current,

    -- Speed
    CHANGE COLUMN speed speed_current DOUBLE NULL,
    ADD COLUMN speed_initial DOUBLE NULL AFTER speed_current,

    -- Distance
    CHANGE COLUMN distance distance_current DOUBLE NULL,
    ADD COLUMN distance_initial DOUBLE NULL AFTER distance_current,

    -- Calorie
    CHANGE COLUMN calorie calories_current DOUBLE NULL,
    ADD COLUMN calories_initial DOUBLE NULL AFTER calories_current,

    -- Vorhandene update_time Spalte zur automatischen Aktualisierung ertüchtigen
    CHANGE COLUMN update_time update_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP;

-- Ursprungswerte initial kopieren
UPDATE tbl_StepTrend 
SET 
    step_count_initial = step_count_current,
    speed_initial = speed_current,
    distance_initial = distance_current,
    calories_initial = calories_current;
